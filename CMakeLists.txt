function(yehenala_enable_warnings target)
    if(MSVC)
        target_compile_options(${target} PRIVATE /W4 /utf-8)
    else()
        target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
        if(MINGW)
            target_compile_options(${target} PRIVATE
                -finput-charset=UTF-8 -fexec-charset=UTF-8)
        endif()
    endif()
endfunction()

# The UI text catalog is scanned from an explicit source list rather than a
# recursive GLOB of the source tree.  A recursive glob here used to pull the
# build trees that live inside the source directory (`build/`,
# `out/build/...`) into the dependency list of the catalog custom command.
# Those paths are regenerated by every build, so the glob result changed on
# every build, CMake re-ran configure each time, and ninja recompiled the whole
# project instead of just the files that had actually changed.  Listing the
# inputs explicitly keeps the dependency set deterministic and independent of
# whatever build trees happen to exist next to the sources.

# Guard against the scan list silently drifting from the project sources: a
# path that no longer exists would otherwise drop its literals from the font
# atlas without any error.  A newly added header that carries user-visible
# strings still has to be added to YEHENALA_CATALOG_HEADERS; the
# ui_text_catalog test fails when a displayed string is missing from the atlas.
foreach(YEHENALA_CATALOG_INPUT IN LISTS YEHENALA_UI_TEXT_INPUTS)
    if(NOT EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/${YEHENALA_CATALOG_INPUT}")
        message(FATAL_ERROR
            "UI text catalog input is missing: ${YEHENALA_CATALOG_INPUT}")
    endif()
endforeach()

set(YEHENALA_REVIEWED_GEOMETRY_SCRIPT
    "${CMAKE_SOURCE_DIR}/tools/build_reviewed_geometry.py")
set(YEHENALA_REVIEWED_GEOMETRY_ADMIN1
    "${CMAKE_SOURCE_DIR}/data/ne_10m_admin_1_states_provinces.geojson")
set(YEHENALA_REVIEWED_GEOMETRY_WATER
    "${CMAKE_SOURCE_DIR}/data/ne_10m_review_water.geojson")
set(YEHENALA_REVIEWED_MAP_RESOURCE
    "${CMAKE_SOURCE_DIR}/data/reviewed_province_geometry.geojson")
set(YEHENALA_REVIEWED_GEOMETRY_STAMP
    "${CMAKE_CURRENT_BINARY_DIR}/reviewed_geometry.stamp")
add_custom_command(
    OUTPUT "${YEHENALA_REVIEWED_GEOMETRY_STAMP}"
    COMMAND ${Python3_EXECUTABLE} "${YEHENALA_REVIEWED_GEOMETRY_SCRIPT}"
            --admin1 "${YEHENALA_REVIEWED_GEOMETRY_ADMIN1}"
            --water "${YEHENALA_REVIEWED_GEOMETRY_WATER}"
            --reviewed "${YEHENALA_REVIEWED_MAP_RESOURCE}"
    # The stamp only records that the generator ran, so it is always refreshed.
    # Change detection lives in the reviewed file: the generator rewrites it
    # only when the bytes differ, and that is what invalidates the embedded map
    # source downstream.  Leaving the stamp untouched whenever the content
    # matched would make it permanently older than the generator inputs and
    # force this eight-second step to re-run on every build.
    COMMAND ${CMAKE_COMMAND} -E touch "${YEHENALA_REVIEWED_GEOMETRY_STAMP}"
    WORKING_DIRECTORY "${CMAKE_SOURCE_DIR}"
    DEPENDS
        "${YEHENALA_REVIEWED_GEOMETRY_SCRIPT}"
        "${YEHENALA_REVIEWED_GEOMETRY_ADMIN1}"
        "${YEHENALA_REVIEWED_GEOMETRY_WATER}"
        "${YEHENALA_REVIEWED_MAP_RESOURCE}"
    VERBATIM
)
add_custom_target(reviewed_geometry
    DEPENDS "${YEHENALA_REVIEWED_GEOMETRY_STAMP}")

# Regenerating the reviewed geometry costs about eleven seconds (GEOS dissolve
# plus the overlap audit over the 38 MB admin-1 source) and it sits at the very
# front of the build graph, so everything that embeds the map waits for it.
# The generated file is deterministic and checked in, which means the local
# inner loop does not need to re-run the generator on every cold build.  Build
# it explicitly with the reviewed_geometry target after editing the whitelist,
# the importer, or the Natural Earth inputs; CI and packaging keep it enabled.
option(YEHENALA_BUILD_REVIEWED_GEOMETRY
    "Regenerate the reviewed province geometry as part of the build" ON)

if(YEHENALA_BUILD_REVIEWED_GEOMETRY)
    # Deliberately an order-only dependency on the executables rather than a
    # dependency of the embedding step.  The generator rewrites the reviewed
    # file only when the bytes differ, and the embedding step already depends on
    # that file, so a real change re-embeds through the normal rule.  Making the
    # embedding wait for the generator instead serialised the first ten seconds
    # of every cold build: the nine-second GEOS pass ran alone while all
    # sixteen cores sat idle, and only then could any translation unit start.
    set(YEHENALA_GEOMETRY_GATE_DEPENDS "${YEHENALA_REVIEWED_GEOMETRY_STAMP}")
else()
    set(YEHENALA_GEOMETRY_GATE_DEPENDS "")
endif()

# The embedded GeoJSON sources are generated, so CMake makes every object of the
# target that owns them wait until they exist.  Keeping them in YehenalaCore
# meant all ~45 core objects waited on the generated map sources, which left the
# first ten seconds of a cold build running barely two steps in parallel.
# Isolating them in their own object library confines that gate to the two
# generated files.
add_library(YehenalaEmbeddedGeoJson OBJECT ${YEHENALA_EMBEDDED_GEOJSON_SOURCES})
target_include_directories(YehenalaEmbeddedGeoJson PRIVATE
    "${CMAKE_CURRENT_SOURCE_DIR}")

# Cross-platform raylib discovery. RAYLIB_ROOT may be supplied explicitly.
if(DEFINED RAYLIB_ROOT AND NOT RAYLIB_ROOT STREQUAL "")
    set(RAYLIB_HINT "${RAYLIB_ROOT}")
else()
    set(RAYLIB_HINT "")
endif()

if(WIN32)
    # Windows uses the static raylib package bundled with the project.
    if(NOT RAYLIB_HINT)
        set(RAYLIB_ROOT "${CMAKE_SOURCE_DIR}/raylib")
    else()
        set(RAYLIB_ROOT "${RAYLIB_HINT}")
    endif()
    message(STATUS "Windows build, using RAYLIB_ROOT=${RAYLIB_ROOT}")
else()
    # Other platforms prefer pkg-config and then search common paths.
    find_package(PkgConfig QUIET)
    if(PKG_CONFIG_FOUND)
        pkg_check_modules(RAYLIB_PC QUIET IMPORTED_TARGET raylib)
    endif()

    if(RAYLIB_PC_FOUND)
        message(STATUS "Found raylib via pkg-config: ${RAYLIB_PC_LIBRARIES}")
    else()
        find_path(RAYLIB_INCLUDE_DIR raylib.h
            PATHS
                ${RAYLIB_HINT}
                ${RAYLIB_HINT}/include
                /usr/local/include
                /usr/include
        )
        find_library(RAYLIB_LIBRARY raylib
            PATHS
                ${RAYLIB_HINT}
                ${RAYLIB_HINT}/lib
                /usr/local/lib
                /usr/lib
        )
        if(RAYLIB_INCLUDE_DIR AND RAYLIB_LIBRARY)
            set(RAYLIB_PC_FOUND TRUE)
            set(RAYLIB_PC_INCLUDE_DIRS ${RAYLIB_INCLUDE_DIR})
            set(RAYLIB_PC_LIBRARIES ${RAYLIB_LIBRARY})
            message(STATUS
                "Found raylib manually: include=${RAYLIB_INCLUDE_DIR}, lib=${RAYLIB_LIBRARY}")
        else()
            message(WARNING
                "raylib not found. Set RAYLIB_ROOT or install raylib.")
        endif()
    endif()
endif()

# Every UI translation unit pulls raylib plus the same foundation headers, and
# re-parsing them per file is the largest compile cost in the project.
# Precompiling that foundation roughly halves the work for the UI targets.
#
# The list is deliberately short.  A precompiled header is invalidated whole,
# so putting a header that gets edited often behind it makes edits to that
# header slower, not faster: including ui_internal.h here measurably pushed a
# one-line change to it from 2s to 14s because all sixteen including files had
# to recompile.  Keeping only headers that rarely change (the standard library
# surface raylib uses, and the value types) preserves the gain without widening
# the edit blast radius.  None of them take part in the input redirection the UI
# tests rely on, so the tests and the application share one generated header.
function(yehenala_use_ui_precompiled_header target)
    target_precompile_headers(${target} PRIVATE
        "<raylib.h>"
        "sim_types.h")
endfunction()

yehenala_use_ui_precompiled_header(YehenalaMarket)

    yehenala_use_ui_precompiled_header(YehenalaConstructionUiTests)
endif()
